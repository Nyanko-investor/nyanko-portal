const options = { timeZone: 'Asia/Tokyo', hour12: false, year: 'numeric', month: '2-digit', day: '2-digit', hour: '2-digit', minute: '2-digit', second: '2-digit' }; const formatter = new Intl.DateTimeFormat('ja-JP', options); const jmaUrl = "https://www.jma.go.jp/bosai/forecast/data/forecast/140000.json"; const meteoUrl = "https://api.open-meteo.com/v1/forecast?latitude=35.4478&longitude=139.6425¤t=relative_humidity_2m&daily=temperature_2m_max,temperature_2m_min&timezone=Asia%2FTokyo"; async function fetchWeather() { try { // ★追加：キャッシュバスター（毎回違うランダムな数字を生成） const uniqueTime = new Date().getTime(); const jmaNoCache = jmaUrl + "?_=" + uniqueTime; const meteoNoCache = meteoUrl + "&_=" + uniqueTime; // URLの後ろにランダム数字をつけて、強制的に最新データを取得 const response = await fetch(jmaNoCache, { cache: "no-store" }); const data = await response.json(); const parts = formatter.formatToParts(new Date()); const year = parseInt(parts.find(p => p.type === 'year').value); const month = parseInt(parts.find(p => p.type === 'month').value); const day = parseInt(parts.find(p => p.type === 'day').value); const h = parts.find(p => p.type === 'hour').value; const m = parts.find(p => p.type === 'minute').value; const dateObj = new Date(year, month - 1, day); const reiwa = year - 2018; const days = ['日', '月', '火', '水', '木', '金', '土']; const dayStr = days[dateObj.getDay()]; document.getElementById('dateText').innerText = `${year}年(令和${reiwa}年) ${month}月${day}日(${dayStr})`; document.getElementById('updateTime').innerText = `${h}:${m} 更新`; let rawWeather = data[0].timeSeries[0].areas[0].weathers[0]; let cleanWeather = rawWeather.replace(/　/g, ' ') .replace(/曇り/g, 'くもり'); document.getElementById('weatherText').innerText = cleanWeather; if (cleanWeather.includes('雨') || cleanWeather.includes('雪')) { if (cleanWeather.includes('一時') || cleanWeather.includes('のち') || cleanWeather.includes('ところにより')) { document.getElementById('umbrellaText').innerText = "折りたたみかさがあると安心！"; } else { document.getElementById('umbrellaText').innerText = "忘れないで持っていこう！"; } } else { document.getElementById('umbrellaText').innerText = "今日は持たなくていいよ！"; } // Open-Meteo側もキャッシュを回避 try { const meteoRes = await fetch(meteoNoCache, { cache: "no-store" }); const meteoData = await meteoRes.json(); const humidity = meteoData.current.relative_humidity_2m; document.getElementById('humidityText').innerText = `(湿度 ${humidity}%)`; const minT = Math.round(meteoData.daily.temperature_2m_min[0]); const maxT = Math.round(meteoData.daily.temperature_2m_max[0]); document.getElementById('minTemp').innerText = minT; document.getElementById('maxTemp').innerText = maxT; let clothesMsg = ""; if (maxT >= 25) { clothesMsg = "半そでがいいね！"; } else if (maxT >= 20) { clothesMsg = "長そでのシャツがピッタリ！"; } else if (maxT >= 15) { clothesMsg = "上着を持っていこう！"; } else if (maxT >= 10) { clothesMsg = "セーターかコートを着てね！"; } else { clothesMsg = "ダウンや手ぶくろで暖かく！"; } document.getElementById('clothesText').innerText = clothesMsg; } catch (e) { document.getElementById('humidityText').innerText = "(湿度 --%)"; document.getElementById('minTemp').innerText = "--"; document.getElementById('maxTemp').innerText = "--"; } } catch (error) { document.getElementById('weatherText').innerText = "エラーが起きたよ"; } } // 初回読み込み fetchWeather(); // タイムワープ検知（5秒ごとに時間を監視し、60秒以上スリープしていたら更新） let lastTime = Date.now(); setInterval(() => { const currentTime = Date.now(); if (currentTime - lastTime > 60000) { fetchWeather(); } lastTime = currentTime; }, 5000); // 念のためのエコ更新（3時間） setTimeout(() => { window.location.reload(); }, 10800000);
